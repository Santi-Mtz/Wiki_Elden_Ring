    \item \textbf{Versión del Sistema}: Android 16 (API 36)
    \item \textbf{Conectividad}: Bluetooth Low Energy (BLE 5.2, modo Peripheral GATT).
\end{itemize}

\subsection{Simulación de Smart TV (Chrome DevTools)}
\begin{itemize}
    \item \textbf{Navegador}: Google Chrome (v126.0)
    \item \textbf{Resolución Configurada}: Fija en \texttt{1920px x 1080px} (Full HD).
    \item \textbf{User Agent Simulado}: \texttt{Mozilla/5.0 (Linux; GoogleTV; Chromecast) AppleWebKit/537.36 (KHTML, like Gecko) Chrome/126.0.0.0 Safari/537.36} (Chrome en Android TV)
    \item \textbf{Safe Zone}: Margen activo de seguridad del 5\% (\texttt{padding: 54px 96px}).
\end{itemize}

\subsection{Resolución de Problemas (Troubleshooting)}
\begin{itemize}
    \item \textbf{Permisos BLE en Android 12+}: API 31+ requiere permisos explícitos de \texttt{BLUETOOTH\_CONNECT} y \texttt{BLUETOOTH\_SCAN}. Solución: Añadidos al \texttt{AndroidManifest.xml}.
    \item \textbf{Crasheo de symlinks en Windows}: Solución: Activación del ``Modo Desarrollador'' en Windows 11.
    \item \textbf{Demora de Render en frío}: Solución: Configuración de fallback automático en el móvil a \texttt{localhost:3000} si la petición excede 4 segundos.
\end{itemize}
